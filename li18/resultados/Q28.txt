SQL EJECUTADO
 SELECT CustomerID, COUNT(DISTINCT InvoiceNo) AS frecuencia_compra, ROUND(SUM(Quantity * UnitPrice), 2) AS
monto_total FROM db3_john.data3_john WHERE Quantity > 0 AND UnitPrice > 0 AND CustomerID IS NOT NULL AND
CustomerID <> '' GROUP BY CustomerID HAVING COUNT(DISTINCT InvoiceNo) >= 5 AND SUM(Quantity * UnitPrice) >=
1000 ORDER BY monto_total DESC;
